  <script>
    // WEB AUDIO API - TỔNG HỢP ÂM THANH
    const audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    function playBeep(type) {
      if (audioCtx.state === 'suspended') audioCtx.resume();
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.connect(gain);
      gain.connect(audioCtx.destination);

      if (type === 'press') {
        osc.frequency.setValueAtTime(450, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.06);
        osc.start(); osc.stop(audioCtx.currentTime + 0.06);
      } else if (type === 'correct') {
        const notes = [523.25, 659.25, 783.99]; // C5, E5, G5
        notes.forEach((freq, idx) => {
          setTimeout(() => {
            const o = audioCtx.createOscillator();
            const g = audioCtx.createGain();
            o.type = 'triangle';
            o.frequency.setValueAtTime(freq, audioCtx.currentTime);
            g.gain.setValueAtTime(0.2, audioCtx.currentTime);
            g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.3);
            o.connect(g); g.connect(audioCtx.destination);
            o.start(); o.stop(audioCtx.currentTime + 0.3);
          }, idx * 80);
        });
      } else if (type === 'wrong') {
        osc.type = 'sawtooth';
        osc.frequency.setValueAtTime(160, audioCtx.currentTime);
        osc.frequency.linearRampToValueAtTime(90, audioCtx.currentTime + 0.25);
        gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.25);
        osc.start(); osc.stop(audioCtx.currentTime + 0.25);
      } else if (type === 'win') {
        const fanfare = [523.25, 659.25, 783.99, 1046.5];
        fanfare.forEach((f, i) => {
          setTimeout(() => {
            const o = audioCtx.createOscillator();
            const g = audioCtx.createGain();
            o.type = 'sine';
            o.frequency.setValueAtTime(f, audioCtx.currentTime);
            g.gain.setValueAtTime(0.25, audioCtx.currentTime);
            g.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5);
            o.connect(g); g.connect(audioCtx.destination);
            o.start(); o.stop(audioCtx.currentTime + 0.5);
          }, i * 140);
        });
      }
    }

    // CẤU HÌNH 8 ẢI
    const STAGE_CONFIG = [
      { id: 1, base: 11, minSub: 2, maxSub: 9, reward: { icon: "🗺️", name: "Bản Đồ Cổ" } },
      { id: 2, base: 12, minSub: 3, maxSub: 9, reward: { icon: "👢", name: "Giày Thần Tốc" } },
      { id: 3, base: 13, minSub: 4, maxSub: 9, reward: { icon: "🛡️", name: "Khiên Dũng Sĩ" } },
      { id: 4, base: 14, minSub: 5, maxSub: 9, reward: { icon: "⚔️", name: "Kiếm Ánh Sáng" } },
      { id: 5, base: 15, minSub: 6, maxSub: 9, reward: { icon: "🦺", name: "Áo Giáp Vàng" } },
      { id: 6, base: 16, minSub: 7, maxSub: 9, reward: { icon: "🧪", name: "Thuốc Thần Kỳ" } },
      { id: 7, base: 17, minSub: 8, maxSub: 9, reward: { icon: "🗝️", name: "Chìa Khóa Vàng" } },
      { id: 8, base: 18, minSub: 9, maxSub: 9, reward: { icon: "👑", name: "Vương Miện Cứu Công Chúa" } }
    ];

    let currentStageIdx = 0;
    let questionsInCurrentStage = [];
    let currentQuestionIdx = 0;
    let currentInput = "";
    let score = 0;
    let wrongAttemptsThisQ = 0;
    let unlockedItems = [];

    // XÁO TRỘN NGẪU NHIÊN CÁC PHÉP TRỪ
    function shuffle(arr) {
      const copy = [...arr];
      for (let i = copy.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }

    function initInventoryUI() {
      const grid = document.getElementById("inventory-grid");
      grid.innerHTML = "";
      STAGE_CONFIG.forEach((cfg, idx) => {
        const itemBox = document.createElement("div");
        itemBox.id = `inv-item-${idx}`;
        itemBox.className = "flex flex-col items-center justify-center p-1.5 rounded-xl border bg-slate-900/60 border-slate-700/80 text-center transition-all";
        itemBox.innerHTML = `
          <div class="text-xl sm:text-2xl opacity-20 filter grayscale">${cfg.reward.icon}</div>
          <div class="text-[9px] text-slate-400 font-bold truncate w-full mt-0.5">Ải ${cfg.id}</div>
        `;
        grid.appendChild(itemBox);
      });
    }

    function updateInventoryUI() {
      STAGE_CONFIG.forEach((cfg, idx) => {
        const el = document.getElementById(`inv-item-${idx}`);
        if (unlockedItems.includes(idx)) {
          el.className = "flex flex-col items-center justify-center p-1.5 rounded-xl border bg-amber-500/20 border-amber-400 text-center shadow-[0_0_10px_rgba(251,191,36,0.3)] animate-pulse";
          el.innerHTML = `
            <div class="text-xl sm:text-2xl">${cfg.reward.icon}</div>
            <div class="text-[9px] text-amber-300 font-bold truncate w-full mt-0.5">${cfg.reward.name}</div>
          `;
        }
      });
    }

    function buildQuestionsForStage(stageIdx) {
      const cfg = STAGE_CONFIG[stageIdx];
      const list = [];
      for (let sub = cfg.minSub; sub <= cfg.maxSub; sub++) {
        list.push({ num1: cfg.base, num2: sub, ans: cfg.base - sub });
      }
      return shuffle(list);
    }

    function loadStage(stageIdx) {
      currentStageIdx = stageIdx;
      questionsInCurrentStage = buildQuestionsForStage(stageIdx);
      currentQuestionIdx = 0;
      wrongAttemptsThisQ = 0;

      const cfg = STAGE_CONFIG[stageIdx];
      document.getElementById("stage-badge").innerText = `Ải ${cfg.id} / 8`;
      document.getElementById("stage-desc").innerText = `ẢI ${cfg.id}: GIẢI PHONG ẤN PHÉP TRỪ ${cfg.base}`;

      loadQuestion();
    }

    function loadQuestion() {
      currentInput = "";
      wrongAttemptsThisQ = 0;
      document.getElementById("wrong-count-box").classList.add("hidden");
      document.getElementById("answer-slot").innerText = "?";
      document.getElementById("answer-slot").classList.remove("text-emerald-400", "text-red-400");
      document.getElementById("answer-slot").classList.add("text-yellow-300");

      const q = questionsInCurrentStage[currentQuestionIdx];
      document.getElementById("math-num1").innerText = q.num1;
      document.getElementById("math-num2").innerText = q.num2;

      const totalQ = questionsInCurrentStage.length;
      document.getElementById("question-progress").innerText = `Phong ấn ${currentQuestionIdx + 1} / ${totalQ}`;
      const percent = (currentQuestionIdx / totalQ) * 100;
      document.getElementById("progress-bar").style.width = `${percent}%`;
    }

    // BÀN PHÍM CẢM ỨNG
    function pressDigit(d) {
      playBeep('press');
      if (currentInput.length >= 2) return;
      currentInput += d.toString();
      document.getElementById("answer-slot").innerText = currentInput;
    }

    function clearDigit() {
      playBeep('press');
      currentInput = "";
      document.getElementById("answer-slot").innerText = "?";
    }

    function submitAnswer() {
      if (currentInput === "") return;
      const q = questionsInCurrentStage[currentQuestionIdx];
      const val = parseInt(currentInput, 10);

      if (val === q.ans) {
        // ĐÚNG
        playBeep('correct');
        score += 2;
        document.getElementById("score-val").innerText = score;
        document.getElementById("answer-slot").classList.remove("text-yellow-300");
        document.getElementById("answer-slot").classList.add("text-emerald-400");

        currentQuestionIdx++;
        if (currentQuestionIdx >= questionsInCurrentStage.length) {
          setTimeout(completeStage, 400);
        } else {
          setTimeout(loadQuestion, 350);
        }
      } else {
        // SAI
        playBeep('wrong');
        wrongAttemptsThisQ++;
        document.getElementById("wrong-count-box").classList.remove("hidden");
        document.getElementById("wrong-val").innerText = wrongAttemptsThisQ;

        const board = document.getElementById("board-container");
        board.classList.add("shake");
        setTimeout(() => board.classList.remove("shake"), 400);

        currentInput = "";
        document.getElementById("answer-slot").innerText = "?";
      }
    }

    function completeStage() {
      unlockedItems.push(currentStageIdx);
      updateInventoryUI();

      const cfg = STAGE_CONFIG[currentStageIdx];

      if (currentStageIdx === STAGE_CONFIG.length - 1) {
        // Màn 8: Giải cứu công chúa thành công
        playBeep('win');
        triggerConfetti();
        document.getElementById("final-score").innerText = score;
        document.getElementById("victory-modal").classList.remove("hidden");
      } else {
        // Nhận bảo vật vượt ải
        playBeep('win');
        document.getElementById("reward-icon").innerText = cfg.reward.icon;
        document.getElementById("reward-title").innerText = cfg.reward.name;
        document.getElementById("reward-desc").innerText = `Xuất sắc! Bạn đã vượt qua Ải ${cfg.id} và thu nạp ${cfg.reward.name} vào hành trang!`;
        document.getElementById("reward-modal").classList.remove("hidden");
      }
    }

    function nextStage() {
      document.getElementById("reward-modal").classList.add("hidden");
      loadStage(currentStageIdx + 1);
    }

    function resetGame() {
      document.getElementById("reward-modal").classList.add("hidden");
      document.getElementById("victory-modal").classList.add("hidden");
      score = 0;
      document.getElementById("score-val").innerText = "0";
      unlockedItems = [];
      initInventoryUI();
      loadStage(0);
    }

    // HIỆU ỨNG PHÁO HOA ĂN MỪNG
    function triggerConfetti() {
      const canvas = document.getElementById("confetti-canvas");
      const ctx = canvas.getContext("2d");
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;

      const particles = [];
      const colors = ["#f59e0b", "#ec4899", "#3b82f6", "#10b981", "#8b5cf6", "#fbbf24"];
      for (let i = 0; i < 120; i++) {
        particles.push({
          x: canvas.width / 2,
          y: canvas.height / 2,
          r: Math.random() * 6 + 3,
          color: colors[Math.floor(Math.random() * colors.length)],
          vx: (Math.random() - 0.5) * 16,
          vy: (Math.random() - 0.5) * 16 - 3,
          gravity: 0.25,
          alpha: 1
        });
      }

      function render() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        let alive = false;
        particles.forEach(p => {
          p.x += p.vx;
          p.y += p.vy;
          p.vy += p.gravity;
          p.alpha -= 0.008;
          if (p.alpha > 0) {
            alive = true;
            ctx.save();
            ctx.globalAlpha = p.alpha;
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
            ctx.fill();
            ctx.restore();
          }
        });
        if (alive) requestAnimationFrame(render);
      }
      render();
    }

    // Khởi động trò chơi
    initInventoryUI();
    loadStage(0);
